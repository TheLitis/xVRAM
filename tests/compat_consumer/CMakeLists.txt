cmake_minimum_required(VERSION 3.25)
project(xvram_compat_profile_consumer LANGUAGES CXX)

if(NOT XVRAM_COMPAT_LIBRARY OR NOT XVRAM_COMPAT_LINK_LIBRARY OR NOT XVRAM_INCLUDE_DIR)
  message(FATAL_ERROR "Compatibility library and header paths are required")
endif()
add_library(compat SHARED IMPORTED)
set_target_properties(compat PROPERTIES IMPORTED_LOCATION "${XVRAM_COMPAT_LIBRARY}")
if(WIN32)
  set_target_properties(compat PROPERTIES IMPORTED_IMPLIB "${XVRAM_COMPAT_LINK_LIBRARY}")
endif()
add_executable(consumer main.cpp)
target_compile_features(consumer PRIVATE cxx_std_20)
target_compile_definitions(consumer PRIVATE XVRAM_USING_CUDA_COMPAT "CONTRACT_${CASE}=1")
target_include_directories(consumer PRIVATE "${XVRAM_INCLUDE_DIR}")
target_include_directories(consumer SYSTEM PRIVATE "${XVRAM_CUDA_INCLUDE_DIR}"
                                                   "${XVRAM_CUBLAS_INCLUDE_DIR}"
                                                   "${XVRAM_CUDA_CRT_HEADERS_DIR}"
                                                   "${XVRAM_CUDA_CCCL_HEADERS_DIR}")
target_link_libraries(consumer PRIVATE compat)
if(MSVC)
  target_compile_options(consumer PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
