cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(xvram_compat_audit_collector LANGUAGES CXX)
  include(CTest)
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The diagnostic CUPTI collector requires a 64-bit process")
endif()
find_package(CUDAToolkit 13.3 REQUIRED)
find_package(Threads REQUIRED)
set(XVRAM_CUPTI_ROOT "${CUDAToolkit_LIBRARY_ROOT}/extras/CUPTI" CACHE PATH "Installed official CUPTI root")
find_path(XVRAM_AUDIT_CUPTI_INCLUDE cupti.h HINTS "${XVRAM_CUPTI_ROOT}/include" REQUIRED)
find_library(XVRAM_AUDIT_CUPTI_LIBRARY NAMES cupti HINTS "${XVRAM_CUPTI_ROOT}/lib64" "${XVRAM_CUPTI_ROOT}/lib" REQUIRED)
add_library(xvram_compat_audit_collector SHARED collector.cpp)
target_compile_features(xvram_compat_audit_collector PRIVATE cxx_std_20)
set_target_properties(xvram_compat_audit_collector PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(xvram_compat_audit_collector SYSTEM PRIVATE "${XVRAM_AUDIT_CUPTI_INCLUDE}" ${CUDAToolkit_INCLUDE_DIRS})
target_link_libraries(xvram_compat_audit_collector PRIVATE "${XVRAM_AUDIT_CUPTI_LIBRARY}" Threads::Threads)
if(COMMAND xvram_enable_warnings)
  xvram_enable_warnings(xvram_compat_audit_collector)
elseif(MSVC)
  target_compile_options(xvram_compat_audit_collector PRIVATE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(xvram_compat_audit_collector PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MSVC)
  target_compile_definitions(xvram_compat_audit_collector PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
option(XVRAM_COMPAT_AUDIT_COLLECTOR_TESTS "Build CPU-only collector core tests" ${BUILD_TESTING})
if(XVRAM_COMPAT_AUDIT_COLLECTOR_TESTS)
  add_executable(collector_core_tests core_tests.cpp)
  target_compile_features(collector_core_tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(collector_core_tests PRIVATE /W4 /WX /permissive- /EHsc)
  else()
    target_compile_options(collector_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME compat_audit_collector_core COMMAND collector_core_tests)
endif()
install(TARGETS xvram_compat_audit_collector RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
install(FILES README.md DESTINATION share/xvram/compat-audit)
