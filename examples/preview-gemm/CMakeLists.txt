cmake_minimum_required(VERSION 3.25)
project(xvram_preview_gemm LANGUAGES CXX)

find_package(xVRAM CONFIG REQUIRED)
if(NOT TARGET xVRAM::cuda_compat)
  message(FATAL_ERROR "Install xVRAM with XVRAM_BUILD_CUDA_COMPAT=ON first")
endif()
add_executable(xvram-preview-gemm main.cpp)
target_compile_features(xvram-preview-gemm PRIVATE cxx_std_20)
target_link_libraries(xvram-preview-gemm PRIVATE xVRAM::cuda_compat)
if(MSVC)
  target_compile_options(xvram-preview-gemm PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(xvram-preview-gemm PRIVATE -Wall -Wextra -Wpedantic -Werror)
  set_target_properties(xvram-preview-gemm PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS xvram-preview-gemm RUNTIME DESTINATION bin)
